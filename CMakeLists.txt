cmake_minimum_required(VERSION 3.24)
project(LocalPong VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_executable(pong_tests game_tests_updated.cpp)
target_include_directories(pong_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME game_rules COMMAND pong_tests)
option(PONG_BUILD_GAME "Build SFML game executables" ON)
if(PONG_BUILD_GAME)
  find_package(SFML 3 REQUIRED CONFIG COMPONENTS Graphics Window System)
  add_executable(pong main_updated.cpp)
  target_link_libraries(pong PRIVATE SFML::Graphics SFML::Window SFML::System)
  add_executable(pong_original EXCLUDE_FROM_ALL main.cpp)
  target_link_libraries(pong_original PRIVATE SFML::Graphics SFML::Window SFML::System)
endif()
